async function exportAsImage() { const btn = document.getElementById('btn-export'); const originalText = btn.innerHTML; btn.innerHTML = ' Mengunduh...'; btn.disabled = true; try { // 1. Pastikan semua font telah selesai dimuatkan await document.fonts.ready; const reportNode = document.getElementById('report-card'); // 2. KEMAS KINI ATRIBUT VALUE PADA SEMUA INPUT (PENTING!) // Ini memastikan html2canvas mendeteksi angka/teks yang diinputkan const inputs = reportNode.querySelectorAll('input'); inputs.forEach(input => { input.setAttribute('value', input.value); }); // 3. Jalankan html2canvas dengan konfigurasi yang betul const canvas = await html2canvas(reportNode, { scale: 3, // Resolusi tinggi useCORS: true, // Benarkan muat turun logo/gambar luaran allowTaint: true, backgroundColor: "#ffffff", logging: false, onclone: (clonedDoc) => { // Pastikan elemen dalam salinan dikemas kini nilainya const clonedInputs = clonedDoc.querySelectorAll('input'); clonedInputs.forEach(input => { input.setAttribute('value', input.value); // Pastikan warna teks kelihatan jelas semasa dieksport input.style.color = "#0f2042"; }); } }); // 4. Muat turun imej PNG const link = document.createElement('a'); link.download = `Laporan_Portrait_BAPAS_Ciangir_${new Date().toISOString().slice(0, 10)}.png`; link.href = canvas.toDataURL('image/png', 1.0); link.click(); } catch (err) { alert("Gagal mengunduh gambar. Silakan coba lagi."); console.error(err); } finally { btn.innerHTML = originalText; btn.disabled = false; } }
